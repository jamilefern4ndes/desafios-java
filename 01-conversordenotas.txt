public class Main {
    public static void main(String[] args) {
        double[] notas = {6.4, 7.0, 7.5, 6.0};
        double soma = 0;
        for(int i = 0; i<notas.length; i++){
            soma = soma+ notas[i];
        }
        double media = soma/notas.length;
        System.out.println("Média completa: " + media);
        int mediaInt = (int)media;
        System.out.println("Média Inteira: " + mediaInt);
    }
}
